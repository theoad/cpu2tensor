cmake_minimum_required(VERSION 3.24)
project(cpu2tensor VERSION 0.5.1 LANGUAGES C CXX)

option(CPU2TENSOR_BUILD_PYTHON "Build the Python extension" OFF)
option(CPU2TENSOR_BUILD_HARDWARE_EXAMPLE "Build the Linux host-kernel trace workload" OFF)
option(CPU2TENSOR_BUILD_KERNEL_EXAMPLE "Build the static Linux guest init example" OFF)
option(CPU2TENSOR_BUILD_WORKER "Build the Linux worker and QEMU plugin" OFF)
set(CPU2TENSOR_QEMU_INCLUDE_DIR "" CACHE PATH "Directory containing the operator's qemu-plugin.h")

add_library(cpu2tensor_core STATIC core/context_filter.cpp core/trace.cpp core/transition_window.cpp)
set_target_properties(cpu2tensor_core PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_include_directories(cpu2tensor_core PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/include")
target_compile_features(cpu2tensor_core PUBLIC cxx_std_20)
target_compile_options(cpu2tensor_core PUBLIC -fno-exceptions -fno-rtti)
target_compile_options(cpu2tensor_core PRIVATE -Wall -Wextra -Wpedantic -Werror)

include(CTest)
if(BUILD_TESTING)
    find_package(Threads REQUIRED)
    add_executable(frame_ring_test tests/frame_ring_test.cpp)
    target_link_libraries(frame_ring_test PRIVATE cpu2tensor_core Threads::Threads)
    target_compile_options(frame_ring_test PRIVATE -UNDEBUG -Wall -Wextra -Wpedantic -Werror)
    add_test(NAME frame_ring COMMAND frame_ring_test)
    add_executable(context_filter_test tests/context_filter_test.cpp)
    target_link_libraries(context_filter_test PRIVATE cpu2tensor_core)
    target_compile_options(context_filter_test PRIVATE -UNDEBUG -Wall -Wextra -Wpedantic -Werror)
    add_test(NAME context_filter COMMAND context_filter_test)
    add_executable(trace_test tests/trace_test.cpp)
    target_link_libraries(trace_test PRIVATE cpu2tensor_core)
    target_compile_options(trace_test PRIVATE -UNDEBUG -Wall -Wextra -Wpedantic -Werror)
    add_test(NAME trace_contract COMMAND trace_test)
    add_executable(transition_window_test tests/transition_window_test.cpp)
    target_link_libraries(transition_window_test PRIVATE cpu2tensor_core Threads::Threads)
    target_compile_options(transition_window_test PRIVATE -UNDEBUG -Wall -Wextra -Wpedantic -Werror)
    add_test(NAME transition_window COMMAND transition_window_test)
endif()

if(CPU2TENSOR_BUILD_HARDWARE_EXAMPLE)
    if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
        message(FATAL_ERROR "The hardware trace workload needs a Linux toolchain.")
    endif()
    find_package(Threads REQUIRED)
    add_executable(hardware_kernel_workload examples/hardware_kernel_workload.c)
    target_compile_features(hardware_kernel_workload PRIVATE c_std_17)
    target_compile_definitions(hardware_kernel_workload PRIVATE _GNU_SOURCE)
    target_compile_options(hardware_kernel_workload PRIVATE -Wall -Wextra -Wpedantic -Werror)

    add_executable(hardware_dirty_pipe_canary examples/hardware_dirty_pipe_canary.c)
    target_compile_features(hardware_dirty_pipe_canary PRIVATE c_std_17)
    target_compile_definitions(hardware_dirty_pipe_canary PRIVATE _GNU_SOURCE)
    target_compile_options(hardware_dirty_pipe_canary PRIVATE -Wall -Wextra -Wpedantic -Werror)
    add_executable(hardware_multimodal_canary examples/hardware_multimodal_canary.c)
    target_compile_features(hardware_multimodal_canary PRIVATE c_std_17)
    target_compile_definitions(hardware_multimodal_canary PRIVATE _GNU_SOURCE)
    target_compile_options(hardware_multimodal_canary PRIVATE -Wall -Wextra -Wpedantic -Werror)
    target_link_libraries(hardware_multimodal_canary PRIVATE Threads::Threads)
    if(BUILD_TESTING)
        function(add_hardware_canary_smoke name mode variant loops)
            add_test(
                NAME ${name}
                COMMAND /bin/sh -c
                    "printf x | \"$1\" \"$2\" \"$3\" \"$4\""
                    canary "$<TARGET_FILE:hardware_multimodal_canary>"
                    "${mode}" "${variant}" "${loops}")
            set_tests_properties(
                ${name} PROPERTIES
                PASS_REGULAR_EXPRESSION
                    "RESULT mode=${mode} variant=${variant} .*oracle=pass"
                TIMEOUT 20)
        endfunction()
        add_hardware_canary_smoke(hardware_canary_flow_ordered flow ordered 128)
        add_hardware_canary_smoke(hardware_canary_flow_permuted flow permuted 128)
        add_hardware_canary_smoke(hardware_canary_memory_hot memory hot 4096)
        add_hardware_canary_smoke(hardware_canary_memory_large memory large 4096)
        add_hardware_canary_smoke(hardware_canary_contention_private contention private 4096)
        add_hardware_canary_smoke(hardware_canary_contention_shared contention shared 4096)
        add_hardware_canary_smoke(hardware_canary_phase_normal phase normal 64)
        add_hardware_canary_smoke(hardware_canary_phase_shifted phase shifted 64)
    endif()
endif()

if(CPU2TENSOR_BUILD_PYTHON)
    find_package(Python REQUIRED COMPONENTS Interpreter Development.Module)
    Python_add_library(_native MODULE WITH_SOABI python/module.cpp)
    target_link_libraries(_native PRIVATE cpu2tensor_core)
    target_compile_options(_native PRIVATE -Wall -Wextra -Wpedantic -Werror)
    install(TARGETS _native LIBRARY DESTINATION cpu2tensor)
endif()

if(CPU2TENSOR_BUILD_WORKER)
    if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
        message(FATAL_ERROR "The QEMU worker needs a Linux toolchain. Use the portable preset on macOS.")
    endif()
    if(NOT EXISTS "${CPU2TENSOR_QEMU_INCLUDE_DIR}/qemu-plugin.h")
        message(FATAL_ERROR "Set CPU2TENSOR_QEMU_INCLUDE_DIR to the matching operator-provided qemu-plugin.h directory. cpu2tensor does not install QEMU.")
    endif()
    find_package(PkgConfig REQUIRED)
    pkg_check_modules(GLIB REQUIRED IMPORTED_TARGET glib-2.0)
    pkg_check_modules(JSONC REQUIRED IMPORTED_TARGET json-c>=0.15)
    find_package(Threads REQUIRED)
    add_library(cpu2tensor_plugin MODULE qemu/plugin.cpp)
    target_include_directories(cpu2tensor_plugin PRIVATE "${CPU2TENSOR_QEMU_INCLUDE_DIR}")
    target_link_libraries(cpu2tensor_plugin PRIVATE cpu2tensor_core PkgConfig::GLIB Threads::Threads ${CMAKE_DL_LIBS})
    target_compile_options(cpu2tensor_plugin PRIVATE -Wall -Wextra -Wpedantic -Werror -fvisibility=hidden)
    add_executable(cpu2tensor-worker worker/main.cpp worker/kernel.cpp)
    target_link_libraries(cpu2tensor-worker PRIVATE cpu2tensor_core PkgConfig::JSONC Threads::Threads)
    target_compile_options(cpu2tensor-worker PRIVATE -Wall -Wextra -Wpedantic -Werror)
    add_executable(checksum examples/checksum.c)
    target_compile_features(checksum PRIVATE c_std_17)
    target_compile_options(checksum PRIVATE -Wall -Wextra -Wpedantic -Werror -fno-pie)
    target_link_options(checksum PRIVATE -no-pie)
    if(CMAKE_SYSTEM_PROCESSOR MATCHES "^(aarch64|arm64)$")
        enable_language(ASM)
        add_executable(layout_paths examples/layout_paths.S)
        target_link_options(layout_paths PRIVATE -nostdlib -static-pie -Wl,--build-id=none)
    endif()
    foreach(example branch_digits stdio_digits)
        add_executable(${example} examples/${example}.c examples/digit_paths.c)
        target_compile_features(${example} PRIVATE c_std_17)
        target_compile_options(${example} PRIVATE -Wall -Wextra -Wpedantic -Werror -fno-pie)
        target_link_options(${example} PRIVATE -no-pie)
    endforeach()
    if(BUILD_TESTING)
        add_executable(kernel_protocol_test tests/kernel_protocol_test.cpp)
        target_include_directories(kernel_protocol_test PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/worker")
        target_link_libraries(kernel_protocol_test PRIVATE cpu2tensor_core PkgConfig::JSONC)
        target_compile_options(kernel_protocol_test PRIVATE -UNDEBUG -Wall -Wextra -Wpedantic -Werror)
        add_test(NAME kernel_protocol COMMAND kernel_protocol_test)
        add_executable(kernel_diagnostics_test tests/kernel_diagnostics_test.cpp worker/kernel.cpp)
        target_include_directories(kernel_diagnostics_test PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/worker")
        target_link_libraries(kernel_diagnostics_test PRIVATE cpu2tensor_core PkgConfig::JSONC Threads::Threads)
        target_compile_options(kernel_diagnostics_test PRIVATE -UNDEBUG -Wall -Wextra -Wpedantic -Werror)
        add_test(NAME kernel_diagnostics COMMAND kernel_diagnostics_test)
        set_tests_properties(kernel_diagnostics PROPERTIES TIMEOUT 35)
        add_executable(window_target tests/window_target.c)
        target_compile_features(window_target PRIVATE c_std_17)
        target_compile_options(window_target PRIVATE -Wall -Wextra -Wpedantic -Werror -fno-pie)
        target_link_options(window_target PRIVATE -no-pie)
        add_library(system_memory_probe MODULE tests/system_memory_probe.cpp)
        target_include_directories(system_memory_probe PRIVATE "${CPU2TENSOR_QEMU_INCLUDE_DIR}")
        target_link_libraries(system_memory_probe PRIVATE cpu2tensor_core PkgConfig::GLIB ${CMAKE_DL_LIBS})
        target_compile_options(system_memory_probe PRIVATE -Wall -Wextra -Wpedantic -Werror)
        add_executable(stdio_test_target tests/stdio_test.c)
        target_compile_features(stdio_test_target PRIVATE c_std_17)
        target_compile_options(stdio_test_target PRIVATE -Wall -Wextra -Wpedantic -Werror)
        add_executable(system_stdio_target tests/system_stdio_target.c)
        target_compile_features(system_stdio_target PRIVATE c_std_17)
        target_compile_options(system_stdio_target PRIVATE -Wall -Wextra -Wpedantic -Werror)
        add_executable(signals_target tests/signals_target.c)
        target_compile_features(signals_target PRIVATE c_std_17)
        target_compile_options(signals_target PRIVATE -Wall -Wextra -Wpedantic -Werror -fno-pie)
        target_link_options(signals_target PRIVATE -no-pie)
        add_executable(plugin_test_target tests/plugin_test.cpp)
        target_link_libraries(plugin_test_target PRIVATE cpu2tensor_core Threads::Threads)
        target_compile_options(plugin_test_target PRIVATE -Wall -Wextra -Wpedantic -Werror)
        add_executable(worker_test_target tests/worker_test_target.cpp)
        target_link_libraries(worker_test_target PRIVATE cpu2tensor_core)
        target_compile_options(worker_test_target PRIVATE -Wall -Wextra -Wpedantic -Werror)
    endif()
endif()
